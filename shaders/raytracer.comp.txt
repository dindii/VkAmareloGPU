uint hash(uint seed) {
    seed = (seed ^ 61u) ^ (seed >> 16u);
    seed *= 9u;
    seed = seed ^ (seed >> 4u);
    seed *= 0x27D4EB2Du;
    seed = seed ^ (seed >> 15u);
    return seed;
}

vec3 sampleRandomVectorWithOffset(uint offset)
{

float randomFloat() { return float(randomUint()) / float(RAND_MAX); }

float randomFloat(float min, float max) {
    return min + (max - min) * randomFloat();

vec3 RayAt(Ray ray, float t)
{
    return ray.Origin + t * ray.Direction;
}

struct HitDetails
{
    vec3 Normal;
    vec3 HitLocation;
    bool FrontFace;
    float t;
    int MaterialType;
    bool Hit;
};

bool IntervalContains(float t, float intervalMin, float intervalMax)
{
    return ( intervalMin <= t && t <= intervalMax);
}

bool IntervalSurrounds(float t, float intervalMin, float intervalMax)
{
    return ( intervalMin < t && t < intervalMax);
}

bool RayHitSphere(Ray ray, Sphere sphere, out HitDetails details)
{
    float t = dot(vec3(sphere.CenterRadius.xyz) - ray.Origin, ray.Direction);
    vec3 point = ray.Origin + ray.Direction * t;
    float height = length(vec3(sphere.CenterRadius.xyz) - point);

    if(height < sphere.CenterRadius.w)
    {
        float x = sqrt(pow(sphere.CenterRadius.w, 2) - pow(height, 2));
        float t1 = t - x;
        float t2 = t + x; 
        
        if(IntervalSurrounds(t1, g_IntervalMin, g_IntervalMax))
        {
            details.t = t1;
        }
        else if(IntervalSurrounds(t2, g_IntervalMin, g_IntervalMax))
        {
            details.t = t2;
        }
        else
        {
            return false;
        }
        
        details.HitLocation = RayAt(ray, details.t);
     //   details.Normal = (details.HitLocation - sphere.Center) / sphere.Radius;
        details.Normal = normalize(details.HitLocation - vec3(sphere.CenterRadius.xyz));
        details.FrontFace = true;

        if(dot(ray.Direction, details.Normal) > 0.0f)
        {
            details.FrontFace = false;
            details.Normal = normalize(vec3(-details.Normal.x, -details.Normal.y, -details.Normal.z));
        }

        return true;
    }

    return false;
}

HitDetails FindClosestHit(Ray ray)
{
    HitDetails closest;
    closest.Hit = false;
    closest.t = g_IntervalMax;

    for(int i = 0; i < ubo[PushConstants.currentFrame].spheresCount; i++)
    {
        HitDetails currentHit;
        if(RayHitSphere(ray, ubo[PushConstants.currentFrame].spheres[i], currentHit))
        {
            if(closest.t > currentHit.t)
            {
                closest = currentHit;
                closest.MaterialType = ubo[PushConstants.currentFrame].spheres[i].MaterialType;
                closest.Hit = true;
            }
        }
    }

    return closest;
}

float Hash(uint x) {
    x ^= x >> 16;
    x *= 0x7feb352d;
    x ^= x >> 15;
    x *= 0x846ca68b;
    x ^= x >> 16;
    return float(x) / float(0xffffffffu);
}

vec3 RandomVec3(uvec3 id, uint seed)
{
    uint h1 = id.x + seed;
    uint h2 = id.y + seed * 1664525u;
    uint h3 = id.z + seed * 1013904223u;
    return vec3(Hash(h1), Hash(h2), Hash(h3));
}

vec3 RandomOnDisk()
{
    vec3 random = sampleRandomVector();
    random.z = 0.0f;
    return random;
}

vec3 RandomOnDiskWithOffset(uint offset)
{
    vec3 random = sampleRandomVectorWithOffset(offset);
    random.z = 0.0f;
    return random;
}

vec3 SampleDisk()
{
    vec3 randomSample = RandomOnDisk();
    return g_CameraCenter + (randomSample.x * g_DefocusDiskU) + (randomSample.y * g_DefocusDiskV);
}

vec3 SampleDiskWithOffset(uint offset)
{
    vec3 randomSample = RandomOnDiskWithOffset(offset);
    return g_CameraCenter + (randomSample.x * g_DefocusDiskU) + (randomSample.y * g_DefocusDiskV);
}


vec3 GetDefaultColor(Ray ray)
{
    return vec3(1.0f, 1.0f, 1.0f);
    vec3 dirNorm = normalize(ray.Direction);
    float alpha = 0.5f * (dirNorm.y + 1.0f);
    return (1.0f - alpha) * vec3(1.0f, 1.0f, 1.0f) + alpha * vec3(0.5f, 0.7f, 1.0f);

    //return texture(RandUnitVecTex, vec2(0.0f, 0.0f)).rgb;
}

vec3 ScatterMetalicBounce(vec3 inDir, vec3 normal, float roughnessFactor, out vec3 biasDirection, uint offsetBase)
{
    vec3 dir = normalize(reflect(inDir, normal)) + (roughnessFactor * sampleRandomVectorWithOffset(offsetBase));
    biasDirection = normal;
    return dir;
}

vec3 ScatterMetalic(vec3 inDir, vec3 normal, float roughnessFactor, out vec3 biasDirection)
{
    vec3 dir = normalize(reflect(inDir, normal)) + (roughnessFactor * sampleRandomVector());
    biasDirection = normal;

        //sometimes  sampleRandomVector(); will generate a vector almost opposite as the normal, in that case, the sum will be almost zero
    //and that would lead us to visual glitches, we check that with a "near zero" check
  
    float epsilon = 1e-8;
    (abs(dir.x) < epsilon && abs(dir.y) < epsilon && abs(dir.z) < epsilon) ? dir = biasDirection : dir = dir;
    dir = normalize(dir);
    return dir;
}

vec3 ScatterLambertian(vec3 normal, out vec3 biasDirection)
{
    vec3 dir = normal + sampleRandomVector();
    biasDirection = normal;

        //sometimes  sampleRandomVector(); will generate a vector almost opposite as the normal, in that case, the sum will be almost zero
    //and that would lead us to visual glitches, we check that with a "near zero" check
    float epsilon = 1e-8;
    (abs(dir.x) < epsilon && abs(dir.y) < epsilon && abs(dir.z) < epsilon) ? dir = biasDirection : dir = dir;
    dir = normalize(dir);

    return dir;
}

float Reflectance(float cosine, float ri)
{
    float r0 = (1.0f - ri) / (1 + ri);
    r0 = r0 * r0;

    return r0 + (1.0f - r0) * pow((1 - cosine), 5);
}

vec3 ScatterDieletric(vec3 inDir, HitDetails hit, float idx, out vec3 biasDirection)
{
    float ri = hit.FrontFace ? (1.0f / idx) : idx;
    vec3 dir = normalize(inDir);
    
    float cosTheta = min(dot(-dir, hit.Normal), 1.0f);
    float sinTheta = sqrt(1.0f - cosTheta*cosTheta);

    vec3 scatteredDir;

    if(ri * sinTheta > 1.0f || Reflectance(cosTheta, ri) > randomFloat())
    {
       biasDirection = hit.Normal;
       scatteredDir = vec3(reflect(inDir, hit.Normal));
    }
    else
    {
        biasDirection = inDir;
        scatteredDir = vec3(refract(inDir, hit.Normal, ri));
    }

    float epsilon = 1e-8;
    (abs(scatteredDir.x) < epsilon && abs(scatteredDir.y) < epsilon && abs(scatteredDir.z) < epsilon) ? scatteredDir = biasDirection : scatteredDir = scatteredDir;

    return scatteredDir;
}

vec3 Scatter(Ray ray, HitDetails hit, out vec3 bias)
{
    Material mat = ubo[PushConstants.currentFrame].materials[hit.MaterialType];

    if(mat.Type == MATERIAL_TYPE_LAMBERTIAN)
    {
        return ScatterLambertian(hit.Normal, bias);
    }
    else if(mat.Type == MATERIAL_TYPE_METAL)
    {
        return ScatterMetalic(ray.Direction, hit.Normal, mat.Roughness, bias);
    }
    else
    {
        return ScatterDieletric(ray.Direction, hit, mat.RefractionIdx, bias);
    }
}

vec3 ResolveColorFromRay(Ray ray)
{
    Ray current = ray;
    vec3 outColor = vec3(1.0f, 1.0f, 1.0f);

    for(int i = 0; i < g_MaxBounces; i++)
    {
        HitDetails hit = FindClosestHit(current);

        if(!hit.Hit)
        {
            return outColor *= GetDefaultColor(current);
        }

        outColor *= ubo[PushConstants.currentFrame].materials[hit.MaterialType].Attenuation.rgb;
        
        vec3 bias;
        
        current.Direction = normalize(Scatter(current, hit, bias));
        current.Origin = hit.HitLocation + bias * 0.0001f;
    }

    return outColor;
}
